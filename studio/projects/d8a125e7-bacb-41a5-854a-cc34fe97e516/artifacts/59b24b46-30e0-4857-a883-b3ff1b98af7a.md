# Product Profile & Skill Extensibility Framework

## Architecture

```
ReplicatedStorage/ProductProfiles/          (shared, domain-agnostic core + data)
├── ProfileRegistry.lua                     CORE — registry of profiles (data-driven)
├── SkillRegistry.lua                       CORE — registry of skills (data-driven)
├── ProjectProfileStore.lua                 CORE — per-project selection persistence + validation
├── DataStoreBackend.lua                    ADAPTER — Roblox DataStore persistence (only Roblox-specific module)
├── MemoryBackend.lua                       ADAPTER — in-memory backend for tests
├── ProfileCatalog.lua                      BOOTSTRAP — registers the 7 built-in profiles
├── Profiles/
│   ├── Roblox.lua                          DATA
│   ├── Web.lua                             DATA
│   ├── Mobile.lua                          DATA
│   ├── AIProduct.lua                       DATA
│   ├── Automation.lua                      DATA
│   ├── Design.lua                          DATA
│   └── Marketing.lua                       DATA
└── ProductProfiles.test.lua                VERIFICATION — executable acceptance tests

Server/ProductProfiles.server.lua           WORKFLOW — wires registry + store + remotes
```

**Design principle:** The three core modules (`ProfileRegistry`, `SkillRegistry`, `ProjectProfileStore`) contain **zero** Roblox API calls and **zero** single-domain branching. They operate purely on data tables and an injected persistence backend. All Roblox-specific behavior is isolated to `DataStoreBackend` and the server workflow script. New disciplines are added as **data modules** registered at runtime — no core code changes.

```lua
--!strict
-- ProfileRegistry.lua
-- CORE (domain-agnostic). Registry of product profiles.
-- Profiles are plain data tables. Registering a new profile requires no
-- modification to this module or any core workflow code.
-- No Roblox API calls. No single-domain branching.

local ProfileRegistry = {}

local profiles: { [string]: any } = {}
local order: { string } = {}

-- Structural validation of a profile data table.
local function validateProfile(profile: any): (boolean, string?)
	if type(profile) ~= "table" then
		return false, "Profile must be a table"
	end
	if type(profile.id) ~= "string" or profile.id == "" then
		return false, "Profile.id must be a non-empty string"
	end
	if type(profile.name) ~= "string" then
		return false, "Profile.name must be a string"
	end
	if type(profile.skills) ~= "table" then
		return false, "Profile.skills must be a table"
	end
	for _, skill in ipairs(profile.skills) do
		if type(skill) ~= "table" then
			return false, "Each skill must be a table"
		end
		if type(skill.id) ~= "string" or skill.id == "" then
			return false, "Skill.id must be a non-empty string"
		end
		if type(skill.name) ~= "string" then
			return false, "Skill.name must be a string"
		end
		if type(skill.tools) ~= "table" then
			return false, "Skill.tools must be a table"
		end
	end
	if type(profile.toolRequirements) ~= "table" then
		return false, "Profile.toolRequirements must be a table"
	end
	return true, nil
end

-- Register a profile. Returns success and an optional error message.
function ProfileRegistry.register(profile: any): (boolean, string?)
	local ok, err = validateProfile(profile)
	if not ok then
		return false, err
	end
	if profiles[profile.id] then
		return false, "Profile already registered: " .. profile.id
	end
	profiles[profile.id] = profile
	table.insert(order, profile.id)
	return true, nil
end

-- Get a registered profile by id (nil if not registered).
function ProfileRegistry.get(id: string): any?
	return profiles[id]
end

-- Resolve a profile id to its profile (alias of get).
function ProfileRegistry.resolve(id: string): any?
	return profiles[id]
end

-- Check whether a profile id is registered.
function ProfileRegistry.isRegistered(id: string): boolean
	return profiles[id] ~= nil
end

-- List all registered profile ids in registration order.
function ProfileRegistry.listIds(): { string }
	local copy: { string } = {}
	for _, id in ipairs(order) do
		table.insert(copy, id)
	end
	return copy
end

-- List all registered profiles in registration order.
function ProfileRegistry.list(): { any }
	local out: { any } = {}
	for _, id in ipairs(order) do
		table.insert(out, profiles[id])
	end
	return out
end

-- Number of registered profiles.
function ProfileRegistry.count(): number
	return #order
end

return ProfileRegistry
```

```lua
--!strict
-- ProjectProfileStore.lua
-- CORE (domain-agnostic). Persists per-project profile selection and validates
-- it against a registry. Depends on an injected persistence backend so it can
-- be tested without a live Roblox DataStore.
-- No Roblox API calls. No single-domain branching.

local ProjectProfileStore = {}

-- Backend interface (injected):
--   backend.get(key: string): string?
--   backend.set(key: string, value: string): (boolean, string?)
--   backend.remove(key: string): (boolean, string?)

local DEFAULT_KEY = "selected_profile"

-- Create a store bound to a registry and a persistence backend.
-- @param registry  must expose isRegistered(id) and resolve(id)
-- @param backend   must expose get/set/remove
-- @param key       optional storage key override
function ProjectProfileStore.new(registry: any, backend: any, key: string?): any
	local storeKey = key or DEFAULT_KEY

	local self = {}

	-- Persist the selected profile id for the current project.
	-- Rejects unknown profile ids (validated against the registry).
	function self.setSelectedProfile(profileId: string): (boolean, string?)
		if type(profileId) ~= "string" then
			return false, "profileId must be a string"
		end
		if not registry.isRegistered(profileId) then
			return false, "Unknown profile: " .. profileId
		end
		local ok, err = backend.set(storeKey, profileId)
		if not ok then
			return false, "Failed to persist selection: " .. tostring(err)
		end
		return true, nil
	end

	-- Load the persisted selection and validate it against the registry.
	-- Returns the profile id if valid, otherwise nil.
	function self.getSelectedProfileId(): string?
		local raw = backend.get(storeKey)
		if not raw or raw == "" then
			return nil
		end
		if not registry.isRegistered(raw) then
			return nil
		end
		return raw
	end

	-- Load the persisted selection and resolve it to the full profile.
	function self.getSelectedProfile(): any?
		local id = self.getSelectedProfileId()
		if not id then
			return nil
		end
		return registry.resolve(id)
	end

	-- Clear the persisted selection.
	function self.clearSelection(): (boolean, string?)
		local ok, err = backend.remove(storeKey)
		if not ok then
			return false, "Failed to clear selection: " .. tostring(err)
		end
		return true, nil
	end

	return self
end

return ProjectProfileStore
```

```lua
-- DataStoreBackend.lua
-- ADAPTER. Roblox DataStore-backed persistence backend for ProjectProfileStore.
-- This is the ONLY module in the framework that touches Roblox persistence APIs.

local DataStoreService = game:GetService("DataStoreService")

local DataStoreBackend = {}

local DEFAULT_STORE_NAME = "ProjectProfileStore"

function DataStoreBackend.new(storeName: string?): any
	local name = storeName or DEFAULT_STORE_NAME
	local store = DataStoreService:GetDataStore(name)

	local self = {}

	function self.get(key: string): string?
		local ok, value = pcall(function()
			return store:GetAsync(key)
		end)
		if ok and type(value) == "string" then
			return value
		end
		return nil
	end

	function self.set(key: string, value: string): (boolean, string?)
		local ok, err = pcall(function()
			store:SetAsync(key, value)
		end)
		if ok then
			return true, nil
		end
		return false, tostring(err)
	end

	function self.remove(key: string): (boolean, string?)
		local ok, err = pcall(function()
			store:RemoveAsync(key)
		end)
		if ok then
			return true, nil
		end
		return false, tostring(err)
	end

	return self
end

return DataStoreBackend
```

```lua
--!strict
-- ProfileCatalog.lua
-- BOOTSTRAP. Registers the built-in product profiles (and their skills) into
-- the registries. This is a data/bootstrap module. Adding a new discipline
-- requires only registering a new data module here (or calling
-- ProfileRegistry.register from anywhere) — no core workflow code changes.

local profiles = {
	require(script.Parent.Profiles.Roblox),
	require(script.Parent.Profiles.Web),
	require(script.Parent.Profiles.Mobile),

	require(script.Parent.Profiles.Automation),
	require(script.Parent.Profiles.Design),
	require(script.Parent.Profiles.Marketing),
```

```lua
--!strict
-- ProductProfiles.server.lua
-- WORKFLOW. Server bootstrap for the Product Profile & Skill framework.
-- Registers built-in profiles, wires the persistence store to a DataStore
-- backend, and exposes server-authoritative remote endpoints for querying
-- and selecting a project profile. Selection is validated against the
-- registry before being persisted.

local ReplicatedStorage = game:GetService("ReplicatedStorage")

local PP = ReplicatedStorage:WaitForChild("ProductProfiles")

local ProfileRegistry = require(PP:WaitForChild("ProfileRegistry"))
local ProfileCatalog = require(PP:WaitForChild("ProfileCatalog"))
local ProjectProfileStore = require(PP:WaitForChild("ProjectProfileStore"))
local DataStoreBackend = require(PP:WaitForChild("DataStoreBackend"))

-- Register all built-in profiles.
local ok, err = ProfileCatalog.registerAll()
if not ok then
	error("[ProductProfiles] Failed to register profiles: " .. tostring(err))
end

-- Create the persistence store bound to the registry and a DataStore backend.
local store = ProjectProfileStore.new(ProfileRegistry, DataStoreBackend.new("ProjectProfileStore"))

-- Remote endpoints.
local remotesFolder = Instance.new("Folder")
remotesFolder.Name = "Remotes"
remotesFolder.Parent = PP

local listProfilesRemote = Instance.new("RemoteFunction")
listProfilesRemote.Name = "ListProfiles"
listProfilesRemote.Parent = remotesFolder

local getSelectedRemote = Instance.new("RemoteFunction")
getSelectedRemote.Name = "GetSelectedProfile"
getSelectedRemote.Parent = remotesFolder

local selectProfileRemote = Instance.new("RemoteEvent")
selectProfileRemote.Name = "SelectProfile"
selectProfileRemote.Parent = remotesFolder

-- Server-authoritative handlers.
listProfilesRemote.OnServerInvoke = function()
	return ProfileRegistry.list()
end

getSelectedRemote.OnServerInvoke = function()
	return store.getSelectedProfile()
end

selectProfileRemote.OnServerEvent:Connect(function(player, profileId)
	if type(profileId) ~= "string" then
		return
	end
	-- Validate against registered profiles before persisting.
	local okSel, errSel = store.setSelectedProfile(profileId)
	if not okSel then
		warn("[ProductProfiles] Rejected selection '" .. tostring(profileId) .. "': " .. tostring(errSel))
	end
end)

print("[ProductProfiles] Framework initialized with " .. tostring(ProfileRegistry.count()) .. " profiles.")
```
