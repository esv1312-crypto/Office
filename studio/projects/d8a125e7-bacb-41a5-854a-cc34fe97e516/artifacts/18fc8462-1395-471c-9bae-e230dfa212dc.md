-- ops/rollout/register.luau
-- Staged rollout and blocker isolation register.
-- Server-only ModuleScript. Persists via DataStoreService.

local DataStoreService = game:GetService("DataStoreService")

local Register = {}
Register.__index = Register

export type Blocker = {
	id: string,
	description: string,
	externalDependency: string,
	impact: string,
	independentWork: string,
	status: "open" | "resolved",
}

export type Stage = {
	id: string,
	name: string,
	entryCriteria: { string },
	exitCriteria: { string },
	deliverables: { string },
	status: "open" | "in_progress" | "complete",
	blockers: { Blocker },
}

export type RegisterState = {
	projectId: string,
	version: number,
	updatedAt: string,
	stages: { Stage },
}

local DEFAULT_STAGES: { Stage } = {

			"QA verdict with evidence.",
		},
		status = "open",
		blockers = {},
	},
}

function Register.new(projectId: string): Register
	local self = setmetatable({}, Register)
	self._projectId = projectId
	self._state = {
		projectId = projectId,
		version = 1,
		updatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ"),
		stages = DEFAULT_STAGES,
	}
	return self
end

function Register:getState(): RegisterState
	return self._state
end

function Register:getStage(stageId: string): Stage?
	for _, stage in ipairs(self._state.stages) do
		if stage.id == stageId then
			return stage
		end
	end
	return nil
end

function Register:setStageStatus(stageId: string, status: "open" | "in_progress" | "complete"): (boolean, string?)
	local stage = self:getStage(stageId)
	if not stage then
		return false, "Stage not found: " .. stageId
	end
	stage.status = status
	self._state.version += 1
	self._state.updatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ")
	return true
end

function Register:addBlocker(stageId: string, blocker: Blocker): (boolean, string?)
	local stage = self:getStage(stageId)
	if not stage then
		return false, "Stage not found: " .. stageId
	end
	table.insert(stage.blockers, blocker)
	self._state.version += 1
	self._state.updatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ")
	return true
end

function Register:updateBlockerStatus(stageId: string, blockerId: string, status: "open" | "resolved"): (boolean, string?)
	local stage = self:getStage(stageId)
	if not stage then
		return false, "Stage not found: " .. stageId
	end
	for _, blocker in ipairs(stage.blockers) do
		if blocker.id == blockerId then
			blocker.status = status
			self._state.version += 1
			self._state.updatedAt = os.date("!%Y-%m-%dT%H:%M:%SZ")
			return true
		end
	end
	return false, "Blocker not found: " .. blockerId
end

-- Serialization: returns a plain table safe for DataStore.
function Register:serialize(): RegisterState
	return self._state
end

-- Deserialization: validates shape and returns a Register or error.
function Register.deserialize(data: any): (Register?, string?)
	if type(data) ~= "table" then
		return nil, "Persisted data is not a table."
	end
	if type(data.projectId) ~= "string" or type(data.version) ~= "number" then
		return nil, "Persisted data missing projectId or version."
	end
	if type(data.stages) ~= "table" then
		return nil, "Persisted data missing stages."
	end
	local reg = Register.new(data.projectId)
	reg._state = data
	return reg
end

-- Persistence (server-only; requires DataStoreService).
local function getStore(projectId: string)
	return DataStoreService:GetDataStore("RolloutRegister", projectId)
end

function Register:persist(): (boolean, string?)
	local ok, err = pcall(function()
		local store = getStore(self._projectId)
		store:SetAsync("current", self:serialize())
	end)
	if not ok then
		return false, tostring(err)
	end
	return true
end

function Register.load(projectId: string): (Register?, string?)
	local store = getStore(projectId)
	local ok, data = pcall(function()
		return store:GetAsync("current")
	end)
	if not ok then
		return nil, tostring(data)
	end
	if data == nil then
		return nil, "No persisted snapshot found for project: " .. projectId
	end
	return Register.deserialize(data)
end

-- Validation against AC1, AC2, AC4. Returns list of issues (empty = pass).
function Register:validate(): { string }
	local issues: { string } = {}

	-- AC1: each stage has entry criteria, exit criteria, deliverables.
	for _, stage in ipairs(self._state.stages) do
		if #stage.entryCriteria == 0 then
			table.insert(issues, "AC1: stage '" .. stage.id .. "' has no entry criteria.")
		end
		if #stage.exitCriteria == 0 then
			table.insert(issues, "AC1: stage '" .. stage.id .. "' has no exit criteria.")
		end
		if #stage.deliverables == 0 then
			table.insert(issues, "AC1: stage '" .. stage.id .. "' has no deliverables.")
		end
	end

	-- AC2: every blocker records external dependency, impact, independent work.
	for _, stage in ipairs(self._state.stages) do
		for _, blocker in ipairs(stage.blockers) do
			if blocker.externalDependency == "" then
				table.insert(issues, "AC2: blocker '" .. blocker.id .. "' missing externalDependency.")
			end
			if blocker.impact == "" then
				table.insert(issues, "AC2: blocker '" .. blocker.id .. "' missing impact.")
			end
			if blocker.independentWork == "" then
				table.insert(issues, "AC2: blocker '" .. blocker.id .. "' missing independentWork.")
			end
		end
	end

	-- AC4: partial work never marked complete; incomplete stages remain visibly open.
	for _, stage in ipairs(self._state.stages) do
		if stage.status == "complete" then
			if #stage.deliverables == 0 then
				table.insert(issues, "AC4: stage '" .. stage.id .. "' marked complete but has no deliverables.")
			end
			for _, blocker in ipairs(stage.blockers) do
				if blocker.status == "open" then
					table.insert(issues, "AC4: stage '" .. stage.id .. "' marked complete but blocker '" .. blocker.id .. "' is open.")
				end
			end
		else
			if stage.status ~= "open" and stage.status ~= "in_progress" then
				table.insert(issues, "AC4: stage '" .. stage.id .. "' has invalid status '" .. tostring(stage.status) .. "'.")
			end
		end
	end

	return issues
end

return Register

--!strict
-- ops/rollout/validate_register.luau
-- Server Script. Loads persisted register, validates AC1-AC4, prints verdict.
-- Run in Studio (Run > Run) or in a live server with DataStore access.

local Register = require(script.Parent.register)

local PROJECT_ID = "d8a125e7-bacb-41a5-854a-cc34fe97e516"

local function main()
	print("=== Rollout Register Validator ===")
	print("Project: " .. PROJECT_ID)

	-- AC3: load persisted snapshot (readback).
	local reg, err = Register.load(PROJECT_ID)
	if not reg then
		print("AC3: FAIL - " .. tostring(err))
		print("VERDICT: revision_required")
		return
	end
	print("AC3: PASS - persisted snapshot loaded (version " .. tostring(reg:getState().version) .. ").")

	-- Round-trip check: serialize and re-deserialize must preserve state.
	local serialized = reg:serialize()
	local reg2, err2 = Register.deserialize(serialized)
	if not reg2 then
		print("AC3: FAIL - round-trip deserialize error: " .. tostring(err2))
		print("VERDICT: revision_required")
		return
	end
	print("AC3: PASS - round-trip serialization preserved state.")

	-- AC1, AC2, AC4 validation.
	local issues = reg:validate()
	if #issues == 0 then
		print("AC1: PASS - all stages have entry criteria, exit criteria, deliverables.")
		print("AC2: PASS - all blockers record external dependency, impact, independent work.")
		print("AC4: PASS - no partial work marked complete; incomplete stages visibly open.")
		print("VERDICT: accepted")
	else
		print("Validation issues found (" .. tostring(#issues) .. "):")
		for _, issue in ipairs(issues) do
			print("  - " .. issue)
		end
		print("VERDICT: revision_required")
	end
end

main()
