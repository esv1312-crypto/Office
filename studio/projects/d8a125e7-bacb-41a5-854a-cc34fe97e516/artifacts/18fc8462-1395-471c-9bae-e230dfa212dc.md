```lua
--!strict
-- ops/rollout/validate_register.luau
-- Mechanical enforcement of AC1, AC2, AC4 against ops/rollout/register.luau.
--
-- Usage (standalone Luau):
--   luau validate_register.luau [path-to-register.luau]
--
-- Exit codes:
--   0 = register is valid, complete, and resumable
--   1 = validation failed (message written to stderr)

local args = { ... }
local REGISTER_PATH = args[1] or "register.luau"

local function fail(message: string)
    io.stderr:write("VALIDATION FAILED: " .. message .. "\n")
    os.exit(1)
end

local function requireNonEmptyStringArray(value: any, what: string, owner: string)
    if type(value) ~= "table" or #value == 0 then
        fail(owner .. " missing " .. what)
    end
    for _, item in ipairs(value) do
        if type(item) ~= "string" or item == "" then
            fail(owner .. " has a non-string or empty entry in " .. what)
        end
    end
end

-- Load and parse the register module.
local chunk, loadErr = loadfile(REGISTER_PATH)
if not chunk then
    fail("register.luau does not parse: " .. tostring(loadErr))
end

local ok, Register = pcall(chunk)
if not ok then
    fail("register.luau threw during load: " .. tostring(Register))
end

if type(Register) ~= "table" then
    fail("register.luau must return a table (module); got " .. type(Register))
end

if type(Register.schemaVersion) ~= "string" or Register.schemaVersion == "" then
    fail("Register.schemaVersion must be a non-empty string")
end

-- AC1: every stage has entry criteria, exit criteria, and deliverables.
if type(Register.stages) ~= "table" or #Register.stages == 0 then
    fail("Register.stages must be a non-empty array")
end

local stageIds = {}
for _, stage in ipairs(Register.stages) do
    if type(stage.id) ~= "string" or stage.id == "" then
        fail("every stage must have a non-empty id")
    end
    if stageIds[stage.id] then
        fail("duplicate stage id: " .. stage.id)
    end
    stageIds[stage.id] = true

    if type(stage.name) ~= "string" or stage.name == "" then
        fail("stage " .. stage.id .. " missing name")
    end
    requireNonEmptyStringArray(stage.entryCriteria, "entryCriteria (AC1)", "stage " .. stage.id)
    requireNonEmptyStringArray(stage.exitCriteria, "exitCriteria (AC1)", "stage " .. stage.id)
    requireNonEmptyStringArray(stage.deliverables, "deliverables (AC1)", "stage " .. stage.id)

    -- AC4: partial work is never marked complete.
    if stage.status == "CLOSED" then
        requireNonEmptyStringArray(stage.evidence, "evidence (AC4)", "stage " .. stage.id)
    elseif stage.status == "OPEN" or stage.status == "IN_PROGRESS" then
        -- Visibly open; acceptable.
    else
        fail("stage " .. stage.id .. " has invalid status: " .. tostring(stage.status))
    end
end

-- Cross-reference check: entry criteria that name stages must name existing stages.
for _, stage in ipairs(Register.stages) do
    for _, criterion in ipairs(stage.entryCriteria) do
        for stageRef in string.gmatch(criterion, "S%d+") do
            if not stageIds[stageRef] then
                fail("stage " .. stage.id .. " entry criterion references unknown stage " .. stageRef)
            end
        end
    end
end

-- AC2: every blocker records external dependency, impact, and independent work.
if type(Register.blockers) ~= "table" or #Register.blockers == 0 then
    fail("Register.blockers must be a non-empty array")
end

local blockerIds = {}
for _, blocker in ipairs(Register.blockers) do
    if type(blocker.id) ~= "string" or blocker.id == "" then
        fail("every blocker must have a non-empty id")
    end
    if blockerIds[blocker.id] then
        fail("duplicate blocker id: " .. blocker.id)
    end
    blockerIds[blocker.id] = true

    if type(blocker.externalDependency) ~= "string" or blocker.externalDependency == "" then
        fail("blocker " .. blocker.id .. " missing externalDependency (AC2)")
    end
    if type(blocker.impact) ~= "string" or blocker.impact == "" then
        fail("blocker " .. blocker.id .. " missing impact (AC2)")
    end
    requireNonEmptyStringArray(blocker.independentWork, "independentWork (AC2)", "blocker " .. blocker.id)
    if type(blocker.status) ~= "string" or blocker.status == "" then
        fail("blocker " .. blocker.id .. " missing status")
    end

    -- Independent work that names stages must name existing stages.
    for _, work in ipairs(blocker.independentWork) do
        for stageRef in string.gmatch(work, "S%d+") do
            if not stageIds[stageRef] then
                fail("blocker " .. blocker.id .. " independent work references unknown stage " .. stageRef)
            end
        end
    end
end

io.write("VALIDATION PASSED: register.luau is a valid, complete, resumable register.\n")
os.exit(0)
```
