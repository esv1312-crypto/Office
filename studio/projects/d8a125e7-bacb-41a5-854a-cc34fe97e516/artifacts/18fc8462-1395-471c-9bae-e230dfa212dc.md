## 1. Status vocabulary and the completion rule

| Status | Meaning | May it be reported as "done"? |
|---|---|---|
| `OPEN` | Not started, or started with no criteria met | No |
| `IN_PROGRESS` | At least one entry criterion met, no exit criterion met | No |
| `BLOCKED` | Work halted on ≥1 open blocker; independent work continues elsewhere | No — remains visibly open |
| `EXIT_MET` | Every exit criterion met **and** each carries ≥1 evidence link | No — pending review sign-off |
| `CLOSED` | `EXIT_MET` + reviewer sign-off recorded + no open blocker attached | Yes |

**Completion rule (normative):**
1. A stage may transition to `EXIT_MET` only when **every** exit criterion has `met: true` and a non-empty `evidence` array.
2. A stage may transition to `CLOSED` only from `EXIT_MET`, with a named reviewer and timestamp, and only if no attached blocker is `OPEN`.
3. Partial work is recorded as `IN_PROGRESS` or `BLOCKED`. There is no "mostly done" status. A stage with 9 of 10 exit criteria met is `IN_PROGRESS`.
4. `BLOCKED` is never a terminal state and never removes the stage from the open list.
5. Any status change without an evidence link is rejected by the validator in §7 (`E_MET_WITHOUT_EVIDENCE`, `E_CLOSED_WITH_OPEN_EXIT`).

**Current register state (seeded, honest):** S0–S7 all `OPEN`; B-01–B-08 all `OPEN`. Zero stages closed. No exit criterion has evidence.

### 5.4 Resume protocol (restart-safe)

1. Read `ops/rollout/state.yaml`; if `persistence.last_known_commit` is `null`, the register has never been committed — stop and commit first.
2. Compare `last_known_commit` to the current repository head. If they differ, read `CHANGELOG.md` for every transition since that SHA before acting.
3. Recompute each stage's status from evidence: a criterion is `met` only if its `evidence` array is non-empty **and** every referenced file exists with the recorded hash. Do not trust the stored `status` field alone.
4. Run the §7 validator. Any error blocks further transitions until fixed.
5. Execute `resume.next_action_id` only. Do not batch multiple transitions in one session.
6. On completion, update `state.yaml`, append one line to `CHANGELOG.md`, and set `persistence.last_known_commit` to the new SHA.

**Restart test (to be executed by an operator, not performed here):** open the register in a fresh session, follow steps 1–4, and confirm the derived `next_action` matches the stored one. If they diverge, the register is not resumable and the divergence is itself a defect.

### File: ops/rollout/RolloutRegisterValidator.luau
```lua
--!strict
-- RolloutRegisterValidator
-- Validates the rollout register state (state.yaml, decoded to a Luau table).
-- Returns a list of error codes; an empty list means the register is consistent.

local Validator = {}

export type Criterion = { id: string, met: boolean, evidence: { string } }

export type Stage = {
	id: string,
	status: string,
	entry_criteria: { Criterion },
	exit_criteria: { Criterion },
	deliverables: { string },
	blockers: { string },
}

export type Blocker = {
	id: string,
	status: string,
	external_dependency: string,
	impact: string,
	blocked_stages: { string },
	continuing_work: { string },
}

export type State = {
	schema_version: number,
	stages: { Stage },
	blockers: { Blocker },
}

local STAGE_STATUS = {
	OPEN = true,
	IN_PROGRESS = true,
	BLOCKED = true,
	EXIT_MET = true,
	CLOSED = true,
}

local BLOCKER_STATUS = {
	OPEN = true,
	MITIGATED = true,
	RESOLVED = true,
}

local function push(errors: { string }, code: string, id: string)
	table.insert(errors, string.format("%s:%s", code, id))
end

function Validator.validate(state: State): { string }
	local errors: { string } = {}

	if state.schema_version ~= 1 then
		push(errors, "E_SCHEMA_VERSION", tostring(state.schema_version))
	end

	local stageById: { [string]: Stage } = {}
	local blockerById: { [string]: Blocker } = {}

	for _, stage in state.stages do
		if stageById[stage.id] ~= nil then
			push(errors, "E_DUPLICATE_STAGE", stage.id)
		end
		stageById[stage.id] = stage
	end

	for _, blocker in state.blockers do
		if blockerById[blocker.id] ~= nil then
			push(errors, "E_DUPLICATE_BLOCKER", blocker.id)
		end
		blockerById[blocker.id] = blocker
	end

	for _, stage in state.stages do
		if not STAGE_STATUS[stage.status] then
			push(errors, "E_STAGE_STATUS", stage.id)
		end
		if #stage.deliverables == 0 then
			push(errors, "E_NO_DELIVERABLES", stage.id)
		end
		if #stage.entry_criteria == 0 or #stage.exit_criteria == 0 then
			push(errors, "E_NO_CRITERIA", stage.id)
		end

		local allExitMet = true
		for _, criterion in stage.exit_criteria do
			if not criterion.met then
				allExitMet = false
			elseif #criterion.evidence == 0 then
				push(errors, "E_MET_WITHOUT_EVIDENCE", stage.id .. "/" .. criterion.id)
			end
		end

		if stage.status == "CLOSED" and not allExitMet then
			push(errors, "E_CLOSED_WITH_OPEN_EXIT", stage.id)
		end
		if stage.status == "EXIT_MET" and not allExitMet then
			push(errors, "E_EXIT_MET_INCOMPLETE", stage.id)
		end
		if stage.status == "BLOCKED" and #stage.blockers == 0 then
			push(errors, "E_BLOCKED_WITHOUT_BLOCKER", stage.id)
		end

		for _, blockerId in stage.blockers do
			local blocker = blockerById[blockerId]
			if blocker == nil then
				push(errors, "E_UNKNOWN_BLOCKER", stage.id .. "/" .. blockerId)
			elseif stage.status == "CLOSED" and blocker.status == "OPEN" then
				push(errors, "E_CLOSED_WITH_OPEN_BLOCKER", stage.id .. "/" .. blockerId)
			end
		end
	end

	for _, blocker in state.blockers do
		if not BLOCKER_STATUS[blocker.status] then
			push(errors, "E_BLOCKER_STATUS", blocker.id)
		end
		if blocker.external_dependency == "" then
			push(errors, "E_NO_DEPENDENCY", blocker.id)
		end
		if blocker.impact == "" then
			push(errors, "E_NO_IMPACT", blocker.id)
		end
		if #blocker.continuing_work == 0 then
			push(errors, "E_NO_CONTINUING_WORK", blocker.id)
		end
		if #blocker.blocked_stages == 0 then
			push(errors, "E_NO_BLOCKED_STAGES", blocker.id)
		end
		for _, stageId in blocker.blocked_stages do
			if stageById[stageId] == nil then
				push(errors, "E_UNKNOWN_STAGE", blocker.id .. "/" .. stageId)
			end
		end
	end

	return errors
end

-- Self-test: builds a minimal valid register, then applies one mutation per
-- invariant and asserts the expected error code appears.
function Validator.selfTest(): (boolean, { string })
	local failures: { string } = {}

	local function criterion(id: string, met: boolean, evidence: { string }?): Criterion
		return { id = id, met = met, evidence = evidence or {} }
	end

	local function baseState(): State
		return {
			schema_version = 1,
			stages = {
				{
					id = "S0",
					status = "OPEN",
					entry_criteria = { criterion("S0-E1", false) },
					exit_criteria = { criterion("S0-X1", false) },
					deliverables = { "D0.1" },
					blockers = {},
				},
			},
			blockers = {
				{
					id = "B-01",
					status = "OPEN",
					external_dependency = "owner grants publish rights",
					impact = "cannot publish",
					blocked_stages = { "S0" },
					continuing_work = { "W1" },
				},
			},
		}
	end

	local function expect(code: string, mutate: (State) -> ())
		local state = baseState()
		mutate(state)
		local errors = Validator.validate(state)
		local found = false
		for _, err in errors do
			if string.sub(err, 1, #code) == code then
				found = true
				break
			end
		end
		if not found then
			table.insert(failures, "expected " .. code .. " but got [" .. table.concat(errors, ", ") .. "]")
		end
	end

	-- Baseline must be clean.
	local baselineErrors = Validator.validate(baseState())
	if #baselineErrors > 0 then
		table.insert(failures, "baseline not clean: " .. table.concat(baselineErrors, ", "))
	end

	expect("E_CLOSED_WITH_OPEN_EXIT", function(s)
		s.stages[1].status = "CLOSED"
	end)

	expect("E_MET_WITHOUT_EVIDENCE", function(s)
		s.stages[1].exit_criteria[1].met = true
	end)

	expect("E_EXIT_MET_INCOMPLETE", function(s)
		s.stages[1].status = "EXIT_MET"
	end)

	expect("E_BLOCKED_WITHOUT_BLOCKER", function(s)
		s.stages[1].status = "BLOCKED"
		s.stages[1].blockers = {}
	end)

	expect("E_NO_CONTINUING_WORK", function(s)
		s.blockers[1].continuing_work = {}
	end)

	expect("E_NO_DEPENDENCY", function(s)
		s.blockers[1].external_dependency = ""
	end)

	expect("E_NO_IMPACT", function(s)
		s.blockers[1].impact = ""
	end)

	expect("E_UNKNOWN_BLOCKER", function(s)
		s.stages[1].blockers = { "B-99" }
	end)

	expect("E_CLOSED_WITH_OPEN_BLOCKER", function(s)
		s.stages[1].status = "CLOSED"
		s.stages[1].exit_criteria[1] = criterion("S0-X1", true, { "evidence/D0.1/report.md" })
		s.stages[1].blockers = { "B-01" }
	end)

	return #failures == 0, failures
end

return Validator
```
