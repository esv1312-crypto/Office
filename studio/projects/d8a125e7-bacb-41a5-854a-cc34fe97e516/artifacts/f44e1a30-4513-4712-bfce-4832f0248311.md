## Implementation Plan

The solution consists of three components:

1. **Server Script** (`LeaderstatsScript.lua`) – Creates a `leaderstats` folder with a `Points` IntValue for each player. This value is replicated to clients, enabling real-time UI updates.
2. **Local Script** (`ScoreDisplayUI.lua`) – Creates a `ScreenGui` containing:
   - A `PointsLabel` displaying the player's current score (top-left).
   - A `LeaderboardFrame` showing the top 10 players (centered, scrollable).
   - Both elements update automatically when `leaderstats` changes.
3. **Optional RemoteEvent Script** (`ScoreEvents.lua`) – Demonstrates how to trigger score changes from other game events.

All UI is created at runtime via `Instance.new` and parented to `player.PlayerGui`, ensuring it appears in the fan plaza.

## File 1: Server Script (`LeaderstatsScript.lua`)

Place this in **ServerScriptService** (or as a ServerScript at the top level). It runs server-side and creates replicated leaderstats.

```lua
-- ServerScript: Leaderstats creation and management
-- Location: ServerScriptService or ReplicatedStorage (as a ModuleScript)
-- Purpose: Creates a 'leaderstats' folder with a 'Points' value for each player.
--          The value is replicated to clients, allowing UI to display and update scores.

local Players = game:GetService("Players")

-- Create leaderstats for each player on join
Players.PlayerAdded:Connect(function(player)
	local leaderstats = Instance.new("Folder")
	leaderstats.Name = "leaderstats"
	leaderstats.Parent = player

	local points = Instance.new("IntValue")
	points.Name = "Points"
	points.Value = 0 -- initial score
	points.Parent = leaderstats
end)

-- Expose a function to add points (can be called by other scripts via require)
local function addPoints(player, amount)
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local points = leaderstats:FindFirstChild("Points")
		if points then
			points.Value = points.Value + amount
		end
	end
end

return {
	addPoints = addPoints,
}
```

-- LocalScript: Score Display and Leaderboard UI
-- Location: StarterPlayer > StarterPlayerScripts (or inserted at runtime)
-- Purpose: Creates a ScreenGui with a score display and a leaderboard.
--          Updates in real-time when the player's Points change.

local Players = game:GetService("Players")
local player = Players.LocalPlayer

-- Wait for the player's character (if needed for positioning context)
-- Not strictly necessary for UI, but ensures the player exists.

-- Create the ScreenGui
local screenGui = Instance.new("ScreenGui")
screenGui.Name = "ScoreDisplayGui"
screenGui.Parent = player.PlayerGui
screenGui.ResetOnSpawn = false -- keep UI across respawns

-- ==================== Score Display (Top-Left) ====================
local scoreFrame = Instance.new("Frame")
scoreFrame.Name = "ScoreFrame"

scoreFrame.Position = UDim2.new(0, 10, 0, 10)

scoreFrame.Parent = screenGui

local scoreLabel = Instance.new("TextLabel")

scoreLabel.Parent = scoreFrame

local leaderboardFrame = Instance.new("Frame")

title.Parent = leaderboardFrame

-- Scrolling frame for the list
local scrollFrame = Instance.new("ScrollingFrame")
scrollFrame.Name = "ScrollFrame"
scrollFrame.Size = UDim2.new(1, -20, 1, -60)
scrollFrame.Position = UDim2.new(0, 10, 0, 50)
scrollFrame.BackgroundTransparency = 1
scrollFrame.ScrollBarThickness = 10
scrollFrame.Parent = leaderboardFrame

-- Layout for the list items
local列表布局 = Instance.new("ListLayout")
列表布局.Name = "ListLayout"
列表布局.Parent = scrollFrame
列表布局.SortOrder = Enum.SortOrder.Numeric
列表Layout.Padding = UDim.new(0, 5)

-- ==================== Data Handling ====================
-- Function to update the score display
local function updateScoreDisplay()
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local points = leaderstats:FindFirstChild("Points")
		if points then
			scoreLabel.Text = "Score: " .. tostring(points.Value)
		end
	end
end

-- Function to build/update the leaderboard
local function updateLeaderboard()
	-- Clear existing entries
	for _, child in ipairs(scrollFrame:GetChildren()) do
		if child.Name ~= "ListLayout" then
			child:Destroy()
		end
	end

	-- Collect all players with leaderstats
	local playersWithScores = {}
	for _, p in ipairs(Players:GetPlayers()) do
		local ls = p:FindFirstChild("leaderstats")
		if ls then
			local pts = ls:FindFirstChild("Points")
			if pts then
				table.insert(playersWithScores, {
					name = p.Name,
					score = pts.Value,
					isSelf = (p == player)
				})
			end
		end
	end

	-- Sort by score descending
	table.sort(playersWithScores, function(a, b)
		return a.score > b.score
	end)

	-- Show top 10
	local maxEntries = math.min(10, #playersWithScores)
	for i = 1, maxEntries do
		local entry = playersWithScores[i]

		-- Row frame
		local row = Instance.new("Frame")
		row.Name = "Entry_" .. i
		row.Size = UDim2.new(1, -10, 0, 40)
		row.BackgroundTransparency = 0.5
		row.backgroundColor = entry.isSelf and Color3.fromRGB(0, 100, 200) or Color3.fromRGB(50, 50, 50)
		row.Parent = scrollFrame

		-- Rank number
		local rankLabel = Instance.new("TextLabel")
		rankLabel.Size = UDim2.new(0, 40, 1, 0)
		rankLabel.Text = tostring(i)
		rankLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
		rankLabel.TextSize = 20
		rankLabel.Font = Enum.Font.Gotham
		rankLabel.BackgroundTransparency = 1
		rankLabel.Parent = row

		-- Player name
		local nameLabel = Instance.new("TextLabel")
		nameLabel.Size = UDim2.new(0.6, -40, 1, 0)
		nameLabel.Position = UDim2.new(0, 40, 0, 0)
		nameLabel.Text = entry.name
		nameLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
		nameLabel.TextSize = 18
		nameLabel.Font = Enum.Font.Gotham
		nameLabel.TextXAlignment = Enum.TextXAlignment.Left
		nameLabel.BackgroundTransparency = 1
		nameLabel.Parent = row

		-- Score
		local scoreLabel = Instance.new("TextLabel")
		scoreLabel.Size = UDim2.new(0.4, 0, 1, 0)
		scoreLabel.Position = UDim2.new(0.6, 0, 0, 0)
		scoreLabel.Text = tostring(entry.score)
		scoreLabel.TextColor3 = Color3.fromRGB(255, 255, 255)
		scoreLabel.TextSize = 18
		scoreLabel.Font = Enum.Font.GothamBold
		scoreLabel.TextXAlignment = Enum.TextXAlignment.Right
		scoreLabel.BackgroundTransparency = 1
		scoreLabel.Parent = row
	end

	-- Adjust canvas size
	scrollFrame.CanvasSize = UDim2.new(0, 0, 0, maxEntries * 45 + (maxEntries - 1) * 5)
end

-- ==================== Event Connections ====================
-- Update score display when the player's Points changes
local function onPlayerPointsChanged(points)
	if points.Name == "Points" then
		updateScoreDisplay()
		updateLeaderboard()
	end
end

-- Connect to player's leaderstats changes
local function setupLeaderboardListener()
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local points = leaderstats:FindFirstChild("Points")
		if points then
			-- Update immediately
			updateScoreDisplay()
			updateLeaderboard()
			-- Listen for changes
			points.Changed:Connect(onPlayerPointsChanged)
		end
	end
end

-- Also listen for new players joining/leaving to update the leaderboard
Players.PlayerAdded:Connect(function()
	updateLeaderboard()
end)

Players.PlayerRemoving:Connect(function()
	updateLeaderboard()
end)

-- Initialize when the player's leaderstats are created (they may already exist)
player.CharacterAdded:Connect(function()
	setupLeaderboardListener()
end)

-- Wait for leaderstats to exist if not already present
local function waitForLeaderstats()
	local leaderstats = player:FindFirstChild("leaderstats")
	while not leaderstats do
		task.wait(0.1)
		leaderstats = player:FindFirstChild("leaderstats")
	end
	setupLeaderboardListener()
end

-- Start the wait coroutine
spawn(waitForLeaderstats)

-- Also update the leaderboard periodically as a fallback (every 2 seconds)
spawn(function()
	while true do
		task.wait(2)
		updateLeaderboard()
	end
end)

## File 3: Optional Score Events Script (`ScoreEvents.lua`)

This script demonstrates how to trigger score changes from other parts of the game. It uses a `RemoteEvent` to safely add points from the client (e.g., when a player touches a part). Place this in **ReplicatedStorage** as a normal script (or use a RemoteEvent in ReplicatedStorage and bind on both sides).

```lua
-- Server Script: ScoreEvents (RemoteEvent handler)
-- Location: ServerScriptService
-- Purpose: Listen for client requests to add points and validate them.

local ReplicatedStorage = game:GetService("ReplicatedStorage")
local Players = game:GetService("Players")

-- Create or get the RemoteEvent
local scoreEvent = ReplicatedStorage:FindFirstChild("ScoreEvent")
if not scoreEvent then
	scoreEvent = Instance.new("RemoteEvent")
	scoreEvent.Name = "ScoreEvent"
	scoreEvent.Parent = ReplicatedStorage
end

-- Handle incoming requests
scoreEvent.OnServerEvent:Connect(function(player, amount)
	-- Validate: only allow integer amounts, limit per request
	if type(amount) ~= "number" then return end
	if amount <= 0 or amount > 1000 then return end -- anti-abuse

	-- Add points to the player
	local leaderstats = player:FindFirstChild("leaderstats")
	if leaderstats then
		local points = leaderstats:FindFirstChild("Points")
		if points then
			points.Value = points.Value + amount
		end
	end
end)
```
