-- Run this in the Developer Console (Ctrl+11) inside Roblox Studio
-- Place in StarterPlayerScripts or execute via a ServerScript

local Players = game:GetService("Players")
local ReplicatedStorage = game:GetService("ReplicatedStorage")

local function testRemoteEvents()
    local events = ReplicatedStorage:GetDescendants()
    local remoteCount = 0
    for _, obj in ipairs(events) do
        if obj:IsA("RemoteEvent") then
            remoteCount = remoteCount + 1
            print(string.format("Found RemoteEvent: %s", obj.Name))
        end
    end
    print(string.format("Total RemoteEvents: %d", remoteCount))
    return remoteCount > 0
end

local function testPlayerSpawn()
    local player = Players.LocalPlayer
    local character = player.Character or player.CharacterAdded:Wait()
    if character then
        print("Player spawned successfully")
        return true
    end
    print("FAIL: Player did not spawn")
    return false
end

-- Execute tests
local eventsOk = testRemoteEvents()
local spawnOk = testPlayerSpawn()

if eventsOk and spawnOk then
    print("BASIC TESTS PASSED")
else
    print("BASIC TESTS FAILED")
end
